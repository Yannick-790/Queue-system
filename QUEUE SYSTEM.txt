// ------------------------------------------------------
// ENUMS
// ------------------------------------------------------

enum Role {
  ADMIN
  MANAGER
  SECURITY
  EMPLOYEE
}

enum RoutingMode {
  FIXED_FLOW
  FLEXIBLE_FLOW
}

enum DeskStatus {
  AVAILABLE
  BUSY
  OFFLINE
}

enum CardStatus {
  AVAILABLE_AT_SECURITY
  WITH_CUSTOMER
  RETURN_PENDING
  LOST
}

enum TicketStatus {
  WAITING
  SERVING
  COMPLETED
  TRANSFERRED
  CANCELLED
  NO_SHOW
}

enum PriorityLevel {
  NORMAL
  URGENT
  EMERGENCY
}

// ------------------------------------------------------
// 1. COMPANIES
// ------------------------------------------------------
model Company {
  id          String        @id @default(uuid())
  name        String
  industry    String?
  routingMode RoutingMode   @default(FIXED_FLOW)
  createdAt   DateTime      @default(now()) @map("created_at")

  buildings   Building[]
  users       User[]
  cards       Card[]
  tickets     Ticket[]
  routes      ServiceRoute[]

  @@map("companies")
}

// ------------------------------------------------------
// 2. BUILDINGS
// ------------------------------------------------------
model Building {
  id          String       @id @default(uuid())
  companyId   String       @map("company_id")
  name        String
  location    String?

  company     Company      @relation(fields: [companyId], references: [id], onDelete: Cascade)
  departments Department[]

  @@map("buildings")
}

// ------------------------------------------------------
// 3. DEPARTMENTS
// ------------------------------------------------------
model Department {
  id          String     @id @default(uuid())
  buildingId  String     @map("building_id")
  name        String

  building    Building   @relation(fields: [buildingId], references: [id], onDelete: Cascade)
  services    Service[]
  desks       Desk[]
  employees   Employee[]

  @@map("departments")
}

// ------------------------------------------------------
// 4. SERVICES
// ------------------------------------------------------
model Service {
  id           String         @id @default(uuid())
  departmentId String         @map("department_id")
  name         String

  department   Department     @relation(fields: [departmentId], references: [id], onDelete: Cascade)
  employees    Employee[]
  tickets      Ticket[]
  
  routesFrom   ServiceRoute[] @relation("FromService")
  routesTo     ServiceRoute[] @relation("ToService")

  @@map("services")
}

// ------------------------------------------------------
// 5. USERS
// ------------------------------------------------------
model User {
  id           String          @id @default(uuid())
  companyId    String          @map("company_id")
  name         String
  email        String          @unique
  passwordHash String          @map("password_hash")
  role         Role            @default(EMPLOYEE)
  createdAt    DateTime        @default(now()) @map("created_at")

  company      Company         @relation(fields: [companyId], references: [id], onDelete: Cascade)
  employee     Employee?
  historyLogs  TicketHistory[]
  notifications Notification[]

  @@map("users")
}

// ------------------------------------------------------
// 6. EMPLOYEES
// ------------------------------------------------------
model Employee {
  id             String           @id @default(uuid())
  userId         String           @unique @map("user_id")
  departmentId   String           @map("department_id")
  serviceId      String?          @map("service_id")
  employeeNumber String?          @map("employee_number")

  user           User             @relation(fields: [userId], references: [id], onDelete: Cascade)
  department     Department       @relation(fields: [departmentId], references: [id], onDelete: Cascade)
  service        Service?         @relation(fields: [serviceId], references: [id], onDelete: SetNull)
  sessions       ServiceSession[]

  @@map("employees")
}

// ------------------------------------------------------
// 7. DESKS
// ------------------------------------------------------
model Desk {
  id           String            @id @default(uuid())
  departmentId String            @map("department_id")
  name         String
  status       DeskStatus        @default(OFFLINE)

  department   Department        @relation(fields: [departmentId], references: [id], onDelete: Cascade)
  sessions     ServiceSession[]

  @@map("desks")
}

// ------------------------------------------------------
// 8. CARDS (Physical Token Stock & Display Mapping)
// ------------------------------------------------------
model Card {
  id            String        @id @default(uuid())
  companyId     String        @map("company_id")
  cardNumber    String        @map("card_number") // External Number shown to customer e.g. "002"
  status        CardStatus    @default(AVAILABLE_AT_SECURITY)

  company       Company       @relation(fields: [companyId], references: [id], onDelete: Cascade)
  tickets       Ticket[]
  cardHistory   CardHistory[]

  @@unique([companyId, cardNumber])
  @@map("cards")
}

// ------------------------------------------------------
// 9. TICKETS (Core Queue Engine)
// ------------------------------------------------------
model Ticket {
  id            String           @id @default(uuid())
  companyId     String           @map("company_id")
  cardId        String           @map("card_id")
  serviceId     String           @map("service_id")
  ticketNumber  String           @map("ticket_number")  // Internal System Sequence ID
  displayNumber String           @map("display_number") // External Number called on TV (e.g. "002")
  priority      PriorityLevel    @default(NORMAL)
  status        TicketStatus     @default(WAITING)
  createdAt     DateTime         @default(now()) @map("created_at")

  company       Company          @relation(fields: [companyId], references: [id], onDelete: Cascade)
  card          Card             @relation(fields: [cardId], references: [id])
  service       Service          @relation(fields: [serviceId], references: [id])
  
  sessions      ServiceSession[]
  history       TicketHistory[]
  cardHistory   CardHistory[]

  @@map("tickets")
}

// ------------------------------------------------------
// 10. SERVICE_SESSIONS (Who served whom & desk timing)
// ------------------------------------------------------
model ServiceSession {
  id         String    @id @default(uuid())
  ticketId   String    @map("ticket_id")
  employeeId String    @map("employee_id")
  deskId     String    @map("desk_id")
  startedAt  DateTime  @default(now()) @map("started_at")
  finishedAt DateTime? @map("finished_at")

  ticket     Ticket    @relation(fields: [ticketId], references: [id], onDelete: Cascade)
  employee   Employee  @relation(fields: [employeeId], references: [id])
  desk       Desk      @relation(fields: [deskId], references: [id])

  @@map("service_sessions")
}

// ------------------------------------------------------
// 11. TICKET_HISTORY (Movement Audit Log)
// ------------------------------------------------------
model TicketHistory {
  id        String   @id @default(uuid())
  ticketId  String   @map("ticket_id")
  userId    String?  @map("user_id")
  action    String
  createdAt DateTime @default(now()) @map("created_at")

  ticket    Ticket   @relation(fields: [ticketId], references: [id], onDelete: Cascade)
  user      User?    @relation(fields: [userId], references: [id], onDelete: SetNull)

  @@map("ticket_history")
}

// ------------------------------------------------------
// 12. CARD_HISTORY (Physical Token Audit Trail)
// ------------------------------------------------------
model CardHistory {
  id        String   @id @default(uuid())
  cardId    String   @map("card_id")
  ticketId  String?  @map("ticket_id")
  action    String
  createdAt DateTime @default(now()) @map("created_at")

  card      Card     @relation(fields: [cardId], references: [id], onDelete: Cascade)
  ticket    Ticket?  @relation(fields: [ticketId], references: [id], onDelete: SetNull)

  @@map("card_history")
}

// ------------------------------------------------------
// 13. SERVICE_ROUTES (Fixed Flow Rules)
// ------------------------------------------------------
model ServiceRoute {
  id            String   @id @default(uuid())
  companyId     String   @map("company_id")
  fromServiceId String   @map("from_service_id")
  toServiceId   String   @map("to_service_id")

  company       Company  @relation(fields: [companyId], references: [id], onDelete: Cascade)
  fromService   Service  @relation("FromService", fields: [fromServiceId], references: [id], onDelete: Cascade)
  toService     Service  @relation("ToService", fields: [toServiceId], references: [id], onDelete: Cascade)

  @@unique([fromServiceId, toServiceId])
  @@map("service_routes")
}

// ------------------------------------------------------
// 14. NOTIFICATIONS (System Alerts)
// ------------------------------------------------------
model Notification {
  id        String   @id @default(uuid())
  userId    String   @map("user_id")
  message   String
  type      String
  status    String   @default("UNREAD")
  createdAt DateTime @default(now()) @map("created_at")

  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@map("notifications")
}
⚡ How This Schema Solves Your exact Setup Requirements
Dual Numbering Built-In:

tickets.ticket_number: Internal sequential ID used for strict queue sorting.

tickets.display_number: The physical card number (002) handed out by security and displayed on main screens/spoken over speakers.

Fixed vs. Flexible Flow Ready:

Uses Company.routingMode (FIXED_FLOW or FLEXIBLE_FLOW) combined with service_routes to automate customer movement.

Card Return Alerts:

When a session finishes, the card status transitions to RETURN_PENDING. A background worker checks for unreturned cards past 5 minutes and pushes a record to the notifications table.

Yes bro. Let me restate your exact bus-station idea, because I understand the flow now.

🚌 Your QueueFlow Bus-Lane Idea

You are not replacing Tap & Go.

You are using the passenger's existing Tap & Go card as an identifier, while QueueFlow manages which passenger belongs to which bus and the order/group in which they board.

1. Bus arrives

Example:

Bus A

Kigali → Huye

The volunteer opens QueueFlow and creates:

Lane A → Bus A → Kigali–Huye

The bus/trip is registered in the system.

2. Passengers come to the lane

Instead of everyone rushing toward the bus, a volunteer controls the lane.

Passenger comes.

Volunteer scans the passenger's Tap & Go card.

QueueFlow records:

Passenger/card ID → Lane A → Bus A

So the volunteer's phone might show:

BUS A — KIGALI → HUYE

001 ✓
002 ✓
003 ✓
004 ✓
005 ✓
006 ✓
...

The passenger doesn't need a new QueueFlow card.

3. QueueFlow creates boarding groups

This is the important part you added.

Instead of requiring perfect order:

001, 002, 003, 004...

you can create groups.

For example:

Group 1: 001–005
Group 2: 006–010
Group 3: 011–015

So if Group 1 is being boarded:

001 ✅
002 ✅
003 ✅
004 ✅
005 ✅

006 ❌ WAIT
007 ❌ WAIT

Once Group 2 opens:

006–010 → ALLOWED

This prevents everybody from pushing toward the bus at once.

4. The volunteer's list goes to the bus

This is the part you clarified.

The volunteer's QueueFlow device has the passenger list.

Before boarding, that list is associated with the bus/trip's scanner/device.

So the bus-side system knows:

These passenger IDs belong to this bus/trip.

Then passengers tap their existing card when boarding.

The system checks:
